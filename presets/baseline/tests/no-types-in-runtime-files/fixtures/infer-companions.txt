import * as v from 'valibot';
import { InferInput } from 'valibot';

export const ConfigSchema = v.object({
  port: v.number(),
  host: v.string(),
});

const LocalSchema = v.object({ ok: v.boolean() });

export type Config = v.InferOutput<typeof ConfigSchema>;
export type ConfigInput = InferInput<typeof ConfigSchema>;
export type Local = v.InferOutput<typeof LocalSchema>;

export function load(raw: object): Config {
  return v.parse(ConfigSchema, raw);
}
